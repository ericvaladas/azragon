# Schwerer Gustav Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the Schwerer Gustav, a gigantic tracked land battleship. From level 1, at most 2 alive, it drives the roads flattening buildings and fires a huge smoke-trailed shell straight at the monster every 10 s. It has 600 HP.

**Architecture:**
- **The unit:** a new `GustavFleet` enemy system modelled on `TankFleet`, with two vertex-coloured instanced meshes (hull, and turret with its gun) built in `gustavModel.ts`.
- **The shell:** a new `gustav` shot kind in `Projectiles`, which flies like a shell, trails smoke and damages like a shell. Enemies fire it through a new `WorldContext.fireGustavShell`.

**Tech Stack:** TypeScript, Three.js r170, Vitest. Run tests with `npx vitest run <file>`, type-check with `npx tsc --noEmit -p .`.

**Spec:** `docs/superpowers/specs/2026-10-16-gustav-design.md`

## Global Constraints

- `CONFIG.spawns.gustav = { unlockLevel: 1, cost: 6, cap: 2, weight: 1 }`.
- `hp` 600 × difficulty `hpMult`; `damage` 60 per shell (`damageMonster` applies difficulty `dmgMult` downstream, as for tank shells), source `'shell'`; one shell every `fireCooldown` 10 s, the first 3–6 s after it first stops; `range` 1100 m; `speed` 25 m/s; `exp` 60; `size` 160 m.
- The shell always hits, like the tanks'. It carries a smoke trail.
- It flattens standing buildings within `crushRadius` as it goes.
- Models face +Z with +Y up; `heading = atan2(dx, dz)`.
- Comments state what code does and its constraints, never history. Don't push or deploy.

## Review Focus

1. **The monster walks out of range mid-fight.** The Gustav must resume driving and must not fire while out of range (Task 3: "drives again when the monster leaves range, and holds fire").
2. **It reaches a target that's moving.** It must re-plan toward the monster's current position (Task 3: "re-plans toward a moving monster").
3. **Killed with a shell in the air.** The shell still lands, and the dead Gustav never fires again (Task 3: "a dead Gustav never fires").
4. **`clear()` mid-run.** No instances may stay drawn (Task 3: "clear removes every instance").
5. **Spine Shield and Reflect.** They see the Gustav shell as an incoming shell, so they can intercept it like any other shell (Task 1: "counts as incoming").

import * as THREE from 'three';
import { mergeGeometries } from 'three/addons/utils/BufferGeometryUtils.js';
import { CONFIG } from '../config';
import { faceted, painted } from '../render/toon';

/**
 * Schwerer Gustav land battleship in units of CONFIG.gustav.size (the instance matrix scales it). Space is +Z forward,
 * +Y up, origin on the ground under the turret pivot. Two meshes: the hull (four track units with road wheels and
 * fenders, side skirts, the armoured hull with its sloped bow, rust weathering, exhaust stacks and stowage on the rear
 * deck) yawed to the heading, and the turret (the box-girder gun carriage with bracing, ladders and vents, the crew
 * platform, the breech, a mantlet with a hazard band, and the enormous tapering barrel) yawed to its aim.
 */
export const GUN = {
  /** Height of the turret ring. */
  ringY: 0.2,
  /** Gun trunnion: height and distance forward of the turret pivot. */
  pivotY: 0.36,
  pivotZ: 0.02,
  /** Barrel elevation (rad) and muzzle distance along the barrel from the trunnion. */
  pitch: 0.08,
  muzzle: 0.92,
} as const;

type Part = THREE.BufferGeometry;
type V3 = [number, number, number];

function box(w: number, h: number, d: number, x: number, y: number, z: number, color: number): Part {
  return painted(faceted(new THREE.BoxGeometry(w, h, d).translate(x, y, z)), color);
}

/** A cylinder whose axis runs along `axis`, centred at (x, y, z). */
function cyl(r: number, h: number, seg: number, axis: 'x' | 'y' | 'z', x: number, y: number, z: number, color: number): Part {
  const g = new THREE.CylinderGeometry(r, r, h, seg);
  if (axis === 'x') g.rotateZ(Math.PI / 2);
  if (axis === 'z') g.rotateX(Math.PI / 2);
  return painted(faceted(g.translate(x, y, z)), color);
}

/** A cylinder along +Z from z0, `len` long, radius r at z0 tapering to rEnd. */
function tube(r: number, len: number, seg: number, z0: number, color: number, rEnd = r): Part {
  return painted(faceted(new THREE.CylinderGeometry(rEnd, r, len, seg).rotateX(Math.PI / 2).translate(0, 0, z0 + len / 2)), color);
}

/** A slab whose side profile is the polygon `zy`, `w` wide across x, centred on x. */
function profile(zy: [number, number][], w: number, color: number): Part {
  const shape = new THREE.Shape(zy.map(([z, y]) => new THREE.Vector2(z, y)));
  // Shape (x, y) extruded along +Z; rotating -90° about Y sends shape x to world z and the extrusion to world -x.
  const g = new THREE.ExtrudeGeometry(shape, { depth: w, bevelEnabled: false }).rotateY(-Math.PI / 2);
  return painted(faceted(g.translate(w / 2, 0, 0)), color);
}

/** A w×h box stretched from `a` to `b` (its local +Z along a→b, local +Y as close to world up as possible). */
function beam(a: V3, b: V3, w: number, h: number, color: number): Part {
  const from = new THREE.Vector3(...a);
  const to = new THREE.Vector3(...b);
  // Matrix4.lookAt(eye, target) points +Z from target to eye.
  const m = new THREE.Matrix4().lookAt(to, from, THREE.Object3D.DEFAULT_UP).setPosition(from.clone().add(to).multiplyScalar(0.5));
  return painted(faceted(new THREE.BoxGeometry(w, h, from.distanceTo(to)).applyMatrix4(m)), color);
}

function merge(parts: Part[]): Part {
  const g = mergeGeometries(parts);
  if (!g) throw new Error('gustavModel: merge failed');
  return g;
}

/** Geometry built along the barrel axis (+Z from the trunnion), raised GUN.pitch and moved onto the trunnion. */
function onAxis(parts: Part[]): Part {
  return merge(parts).rotateX(-GUN.pitch).translate(0, GUN.pivotY, GUN.pivotZ);
}

/** Four track units (two per side, front and rear), side skirts, the armoured hull and its rear deck. */
export function hullGeometry(): Part {
  const P = CONFIG.palette;
  const parts: Part[] = [];
  for (const sx of [-1, 1]) {
    const x = sx * 0.19;
    for (const zc of [0.25, -0.25]) {
      parts.push(profile([[zc - 0.22, 0.07], [zc - 0.18, 0], [zc + 0.18, 0], [zc + 0.22, 0.07], [zc + 0.2, 0.1], [zc - 0.2, 0.1]], 0.1, P.gustavTrack).translate(x, 0, 0));
      for (const dz of [-0.15, -0.05, 0.05, 0.15]) parts.push(cyl(0.035, 0.012, 8, 'x', sx * 0.245, 0.045, zc + dz, P.gustavMetal));
      parts.push(box(0.12, 0.015, 0.46, x, 0.108, zc, P.gustav));
    }
    parts.push(box(0.02, 0.05, 0.9, sx * 0.25, 0.09, 0, P.gustavDark));
  }
  parts.push(box(0.3, 0.08, 0.96, 0, 0.09, 0, P.gustavDark));
  // Upper hull: a flat deck running into a sloped bow, and a short slope down at the stern.
  parts.push(profile([[-0.5, 0.12], [-0.5, 0.19], [-0.46, 0.21], [0.34, 0.21], [0.52, 0.14], [0.5, 0.12]], 0.44, P.gustav));
  // Rust weathering on the deck.
  parts.push(box(0.16, 0.006, 0.09, 0.1, 0.212, -0.3, P.gustavRust));
  parts.push(box(0.1, 0.006, 0.07, -0.12, 0.212, 0.24, P.gustavRust));
  // Rear deck: twin exhaust stacks and a stowage bin.
  for (const sx of [-1, 1]) parts.push(cyl(0.025, 0.1, 8, 'y', sx * 0.1, 0.26, -0.42, P.gustavMetal));
  parts.push(box(0.3, 0.03, 0.08, 0, 0.225, -0.34, P.gustavDark));
  return merge(parts);
}

/** The turret ring, the box-girder carriage, the crew platform and the gun. */
export function turretGeometry(): Part {
  const P = CONFIG.palette;
  const parts: Part[] = [cyl(0.16, 0.03, 12, 'y', 0, GUN.ringY + 0.015, 0, P.gustavDark)];
  for (const sx of [-1, 1]) {
    const x = sx * 0.11;
    const out = x + sx * 0.018;
    // A tall girder wall with X-bracing, a ladder and vents on its outer face, and a rust streak.
    parts.push(box(0.03, 0.2, 0.4, x, 0.33, -0.02, P.gustav));
    parts.push(beam([out, 0.25, -0.2], [out, 0.41, 0.16], 0.008, 0.02, P.gustavDark));
    parts.push(beam([out, 0.41, -0.2], [out, 0.25, 0.16], 0.008, 0.02, P.gustavDark));
    parts.push(box(0.006, 0.18, 0.03, x + sx * 0.02, 0.32, -0.18, P.gustavMetal));
    for (let k = 0; k < 3; k++) parts.push(box(0.006, 0.015, 0.05, out, 0.39 - k * 0.03, -0.08, P.gustavDark));
    parts.push(box(0.006, 0.08, 0.02, out, 0.3, 0.1, P.gustavRust));
  }
  // Crew platform behind the breech, with a rail.
  parts.push(box(0.26, 0.02, 0.18, 0, 0.44, -0.12, P.gustavDark));
  parts.push(box(0.26, 0.04, 0.012, 0, 0.47, -0.21, P.gustavMetal));
  const band: Part[] = [];
  for (let k = 0; k < 6; k++) band.push(box(0.03, 0.165, 0.055, -0.075 + k * 0.03, 0, 0.02, k % 2 ? P.gustavHazardDark : P.gustavHazard));
  parts.push(
    onAxis([
      box(0.16, 0.14, 0.2, 0, 0, -0.12, P.gustavDark),
      box(0.18, 0.16, 0.05, 0, 0, 0.02, P.gustav),
      ...band,
      tube(0.045, GUN.muzzle - 0.09, 10, 0.05, P.gustav, 0.03),
      tube(0.04, 0.04, 10, GUN.muzzle - 0.04, P.gustavDark),
    ]),
  );
  return merge(parts);
}

/** Muzzle of a Gustav of `size` with its turret yawed `yaw`, relative to its ground origin (world units). */
export function gustavMuzzle(yaw: number, size: number, out: THREE.Vector3): THREE.Vector3 {
  const c = Math.cos(GUN.pitch);
  const along = GUN.pivotZ + GUN.muzzle * c;
  return out.set(Math.sin(yaw) * along, GUN.pivotY + GUN.muzzle * Math.sin(GUN.pitch), Math.cos(yaw) * along).multiplyScalar(size);
}

import * as THREE from 'three';
import { CONFIG } from '../config';
import type { Target } from '../combat';
import type { EnemySystem, WorldContext } from './types';
import { toonMat } from '../render/toon';
import { angleDelta, dist2, type Vec2 } from '../util/math';
import { gustavMuzzle, hullGeometry, turretGeometry } from './gustavModel';

const G = CONFIG.gustav;
const MAX = CONFIG.spawns.gustav.cap;
const _m = new THREE.Matrix4();
const _q = new THREE.Quaternion();
const _p = new THREE.Vector3();
const _s = new THREE.Vector3();
const _c = new THREE.Color();
const _v = new THREE.Vector3();
const UP = new THREE.Vector3(0, 1, 0);

/** Turns `from` toward `to` by at most `step` rad. */
const turn = (from: number, to: number, step: number) => from + Math.max(-step, Math.min(step, angleDelta(from, to)));

class Gustav implements Target {
  readonly kind = 'gustav' as const;
  readonly airborne = false;
  readonly radius = G.size * 0.45;
  readonly position = new THREE.Vector3();
  hp: number = G.hp;
  alive = true;
  heading = 0;
  turretYaw = 0;
  path: Vec2[] = [];
  replanT = 0;
  /** Within range of the monster (stopped) since the last time it drove, and the time to its next shot. */
  engaged = false;
  fireCd = 0;
  flashT = 0;

  constructor(private onDamage: (g: Gustav, amount: number) => void) {}

  isAlive(): boolean {
    return this.alive;
  }

  applyDamage(amount: number): void {
    this.onDamage(this, amount);
  }
}

/**
 * Schwerer Gustav land battleships (at most CONFIG.spawns.gustav.cap): each follows the roads toward the monster,
 * flattening the buildings it rolls into, and within range stops, swings its gun round and fires a huge shell
 * straight at the monster on a slow reload. Drawn as two vertex-coloured InstancedMeshes (hull; turret with gun).
 */
export class GustavFleet implements EnemySystem {
  readonly kind = 'gustav' as const;
  private ctx: WorldContext | null = null;
  private units: Gustav[] = [];
  private hull: THREE.InstancedMesh;
  private turret: THREE.InstancedMesh;

  constructor(scene: THREE.Scene) {
    const mat = toonMat(0xffffff, { vertexColors: true });
    const mk = (geo: THREE.BufferGeometry) => {
      const m = new THREE.InstancedMesh(geo, mat, MAX);
      m.castShadow = true;
      m.count = 0;
      m.frustumCulled = false;
      for (let i = 0; i < MAX; i++) m.setColorAt(i, _c.setHex(0xffffff));
      scene.add(m);
      return m;
    };
    this.hull = mk(hullGeometry());
    this.turret = mk(turretGeometry());
  }

  targets(): readonly Target[] {
    return this.units;
  }

  count(): number {
    return this.units.length;
  }

  clear(): void {
    this.units = [];
    this.draw();
  }

  /** On the road nearest a random point of the spawn ring; false at the cap. */
  spawn(ctx: WorldContext): boolean {
    const a = ctx.rng() * Math.PI * 2;
    const r = CONFIG.director.spawnRing * (0.8 + ctx.rng() * 0.3);
    const m = ctx.monster.position;
    const p = ctx.roads.nearest({ x: m.x + Math.cos(a) * r, z: m.z + Math.sin(a) * r });
    return this.spawnAt(p.x, p.z, ctx);
  }

  /** A Gustav at an exact point (assumed on a road), facing the monster, HP scaled by difficulty; false at the cap. */
  spawnAt(x: number, z: number, ctx: WorldContext): boolean {
    this.ctx = ctx;
    if (this.units.length >= MAX) return false;
    const g = new Gustav(this.damage);
    g.hp = G.hp * ctx.difficulty.hpMult;
    g.position.set(x, G.size * 0.2, z);
    const m = ctx.monster.position;
    g.heading = g.turretYaw = Math.atan2(m.x - x, m.z - z);
    this.units.push(g);
    this.draw();
    return true;
  }

  private damage = (g: Gustav, amount: number): void => {
    if (!g.alive) return;
    g.hp -= amount;
    g.flashT = 0.1;
    if (g.hp > 0) return;
    g.alive = false;
    const ctx = this.ctx;
    if (!ctx) return;
    // A huge blast: several fireballs over the hull, a dust ring, and the buildings around it come down.
    for (const k of [-0.3, 0, 0.3]) {
      _v.set(g.position.x + Math.sin(g.heading) * G.size * k, g.position.y, g.position.z + Math.cos(g.heading) * G.size * k);
      ctx.particles.explosion(_v, 2.2);
    }
    ctx.particles.ring('dust', _v.set(g.position.x, 0, g.position.z), G.deathBlastRadius, 24);
    ctx.damageBuildings({ x: g.position.x, z: g.position.z }, G.deathBlastRadius, G.deathBlastDamage);
    ctx.shake(0.6);
    ctx.onDestroyed(g, G.exp);
  };

  update(dt: number, ctx: WorldContext): void {
    this.ctx = ctx;
    const m = ctx.monster.position;
    const goal = { x: m.x, z: m.z };
    for (let i = this.units.length - 1; i >= 0; i--) {
      const g = this.units[i]!;
      if (!g.alive) {
        this.units.splice(i, 1);
        continue;
      }
      g.flashT = Math.max(0, g.flashT - dt);
      const here = { x: g.position.x, z: g.position.z };
      ctx.damageBuildings(here, G.crushRadius, G.crushDps * dt);
      const aim = Math.atan2(goal.x - here.x, goal.z - here.z);
      g.turretYaw = turn(g.turretYaw, aim, G.aimRate * dt);
      if (dist2(here, goal) <= G.range) {
        if (!g.engaged) {
          g.engaged = true;
          g.fireCd = G.firstShotMin + ctx.rng() * (G.firstShotMax - G.firstShotMin);
        }
        if (Math.abs(angleDelta(g.turretYaw, aim)) > G.aimTolerance) continue;
        g.fireCd -= dt;
        if (g.fireCd <= 0) {
          g.fireCd = G.fireCooldown;
          this.fire(g, ctx);
        }
        continue;
      }
      g.engaged = false;
      g.replanT -= dt;
      if (g.path.length === 0 || g.replanT <= 0) {
        g.path = ctx.roads.path(here, goal);
        g.replanT = G.replanEvery;
      }
      const wp = g.path[0];
      if (!wp) continue;
      const wd = dist2(here, wp);
      const step = Math.min(wd, G.speed * dt);
      if (wd > 1e-3) {
        g.heading = turn(g.heading, Math.atan2(wp.x - here.x, wp.z - here.z), G.turnRate * dt);
        g.position.x += ((wp.x - here.x) / wd) * step;
        g.position.z += ((wp.z - here.z) / wd) * step;
      }
      if (wd - step < 1) g.path.shift();
    }
    this.draw();
  }

  /** One shell from the muzzle at the monster's chest, with a big muzzle blast and a shake. */
  private fire(g: Gustav, ctx: WorldContext): void {
    const muzzle = gustavMuzzle(g.turretYaw, G.size, new THREE.Vector3()).add(_v.set(g.position.x, 0, g.position.z));
    const target = ctx.monster.position.clone().setY(ctx.monster.position.y + ctx.monster.height * 0.55);
    ctx.fireGustavShell(muzzle, target, G.damage);
    ctx.particles.emit('fire', muzzle, 10, 1.2);
    ctx.particles.emit('smoke', muzzle, 8, 1.5);
    ctx.shake(G.fireShake);
  }

  private draw(): void {
    this.units.forEach((g, i) => {
      _p.set(g.position.x, 0, g.position.z);
      _s.setScalar(G.size);
      _m.compose(_p, _q.setFromAxisAngle(UP, g.heading), _s);
      this.hull.setMatrixAt(i, _m);
      _m.compose(_p, _q.setFromAxisAngle(UP, g.turretYaw), _s);
      this.turret.setMatrixAt(i, _m);
      _c.setHex(g.flashT > 0 ? 0xffd0a0 : 0xffffff);
      this.hull.setColorAt(i, _c);
      this.turret.setColorAt(i, _c);
    });
    for (const m of [this.hull, this.turret]) {
      m.count = this.units.length;
      m.instanceMatrix.needsUpdate = true;
      if (m.instanceColor) m.instanceColor.needsUpdate = true;
    }
  }
}
